 $("#searchExpense").keyup(function () {

        let value = $(this).val().toLowerCase();

        $("#expenseTable tr").filter(function () {

            $(this).toggle(

                $(this).text().toLowerCase().indexOf(value) > -1

            );

        });

    });


    // ================= SORT =================

    $("#sortExpense").change(function () {

        let option = $(this).val();

        if (option == "low") {

            expenses.sort(function (a, b) {

                return a.amount - b.amount;

            });

        }

        else if (option == "high") {

            expenses.sort(function (a, b) {

                return b.amount - a.amount;

            });

        }

        else if (option == "new") {

            expenses.sort(function (a, b) {

                return new Date(b.date) - new Date(a.date);

            });

        }

        else if (option == "old") {

            expenses.sort(function (a, b) {

                return new Date(a.date) - new Date(b.date);

            });

        }

        saveData();

    });

$(document).ready(function () {

    let expenses = [];
    let editIndex = -1;

    $("#expenseForm").submit(function (para) {
        para.preventDefault();

        let name = $("#expenseName").val();
        let amount = $("#expenseAmount").val();
        let date = $("#expenseDate").val();
        let category = $("#expenseCategory").val();
        let payment = $("#paymentMethod").val();
        // adding 
        if (editIndex == -1) {

            expenses.push({
                name: name,
                amount: amount,
                date: date,
                category: category,
                payment: payment
            });
        // updating
        } else {

            expenses[editIndex].name = name;
            expenses[editIndex].amount = amount;
            expenses[editIndex].date = date;
            expenses[editIndex].category = category;
            expenses[editIndex].payment = payment;

            editIndex = -1;
            $("button[type='submit']").text("Add Expense");
        }

        displayExpenses();

        $("#expenseForm")[0].reset();
    });


    // DISPLAY EXPENSES hero 
function displayExpenses() {

    $("#expenseTable").html("");

    let total = 0;
    let highest = 0;
    let monthTotal = 0;

    let currentMonth = new Date().getMonth();
    let currentYear = new Date().getFullYear();

    for (let i = 0; i < expenses.length; i++) {

        total = total + Number(expenses[i].amount);

        // Highest Expense
        if (Number(expenses[i].amount) > highest) {
            highest = Number(expenses[i].amount);
        }

        // This Month Expense
        let d = new Date(expenses[i].date);

        if (d.getMonth() == currentMonth &&
            d.getFullYear() == currentYear) {

            monthTotal = monthTotal + Number(expenses[i].amount);
        }

        $("#expenseTable").append(`
            <tr>
                <td>${expenses[i].date}</td>
                <td>${expenses[i].category}</td>
                <td>₹${expenses[i].amount}</td>
                <td>${expenses[i].payment}</td>
                <td>${expenses[i].name}</td>
                <td>
                    <button class="editBtn" data-index="${i}">Edit</button>
                    <button class="deleteBtn" data-index="${i}">Delete</button>
                </td>
            </tr>
        `);
    }

    let average = 0;

    if (expenses.length > 0) {
        average = total / expenses.length;
    }

    $("#totalExpense").text("₹" + total);
    $("#avgExpense").text("₹" + average.toFixed(2));
    $("#highestExpense").text("₹" + highest);
    $("#thisMonth").text("₹" + monthTotal);
    $("#totalEntries").text(expenses.length);
}
    $(document).on("click", ".deleteBtn", function () {

        let index = $(this).data("index");

        for (let i = index; i < expenses.length - 1; i++) {
            expenses[i] = expenses[i + 1];
        }

        expenses.length--;

        displayExpenses();
    });

    $(document).on("click", ".editBtn", function () {

        let index = $(this).data("index");

        $("#expenseName").val(expenses[index].name);
        $("#expenseAmount").val(expenses[index].amount);
        $("#expenseDate").val(expenses[index].date);
        $("#expenseCategory").val(expenses[index].category);
        $("#paymentMethod").val(expenses[index].payment);

        editIndex = index;

        $("button[type='submit']").text("Update Expense");
    });

});

$(document).ready(function () {

    let expenses = JSON.parse(localStorage.getItem("expenses")) || [];

    let editId = null;

    displayExpenses();

    // ================= ADD / UPDATE feature===================================

    $("#expenseForm").submit(function (e) {

        e.preventDefault();

        let expense = {

            id: editId ? editId : Date.now(),

            name: $("#expenseName").val(),

            amount: Number($("#expenseAmount").val()),

            date: $("#expenseDate").val(),

            category: $("#expenseCategory").val(),

            payment: $("#paymentMethod").val()

        };

        if (editId == null) {

            expenses.push(expense);

        }

        else {

            expenses = expenses.map(function (item) {

                if (item.id == editId) {

                    return expense;

                }

                return item;

            });

            editId = null;

            $("button[type='submit']").text("Add Expense");

        }

        saveData();

        $("#expenseForm")[0].reset();

    });


    // ================= SAVE =====================================

    function saveData() {

        localStorage.setItem("expenses", JSON.stringify(expenses));

        displayExpenses();

    }


    // ================= DISPLAY ==================================================================

    function displayExpenses() {

        $("#expenseTable").html("");

        let total = 0;

        let highest = 0;

        let monthTotal = 0;

        let currentMonth = new Date().getMonth();

        let currentYear = new Date().getFullYear();

        $.each(expenses, function (index, item) {

            total += item.amount;

            if (item.amount > highest) {

                highest = item.amount;

            }

            let d = new Date(item.date);

            if (d.getMonth() == currentMonth &&
                d.getFullYear() == currentYear) {

                monthTotal += item.amount;

            }

            $("#expenseTable").append(`
                <tr>
                    <td>${item.date}</td>
                    <td>${item.category}</td>
                    <td>₹${item.amount}</td>
                    <td>${item.payment}</td>
                    <td>${item.name}</td>

                    <td>

                        <button class="btn btn-warning btn-sm editBtn"
                            data-id="${item.id}">
                            Edit
                        </button>

                        <button class="btn btn-danger btn-sm deleteBtn"
                            data-id="${item.id}">
                            Delete
                        </button>

                    </td>

                </tr>
            `);

        });

        let average = 0;

        if (expenses.length > 0) {

            average = total / expenses.length;

        }

        $("#totalExpense").text("₹" + total);

        $("#avgExpense").text("₹" + average.toFixed(2));

        $("#highestExpense").text("₹" + highest);

        $("#thisMonth").text("₹" + monthTotal);

        $("#totalEntries").text(expenses.length);

    }


    // ================= DELETE =================

    $(document).on("click", ".deleteBtn", function () {

        let id = $(this).data("id");

        expenses = expenses.filter(function (item) {

            return item.id != id;

        });

        saveData();

    });


    // ================= EDIT =================

    $(document).on("click", ".editBtn", function () {

        let id = $(this).data("id");

        let expense = expenses.find(function (item) {

            return item.id == id;

        });

        $("#expenseName").val(expense.name);

        $("#expenseAmount").val(expense.amount);

        $("#expenseDate").val(expense.date);

        $("#expenseCategory").val(expense.category);

        $("#paymentMethod").val(expense.payment);

        editId = id;

        $("button[type='submit']").text("Update Expense");

    });


    // ================= SEARCH and Sorting =================

   
});
